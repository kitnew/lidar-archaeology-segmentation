\newglossaryentry{hillshade}{
    name={Hillshade},
    description={is a terrain visualization that simulates illumination from a selected direction in order to emphasize relief forms}
}

\newglossaryentry{slope}{
    name={Slope},
    description={is a terrain derivative describing the local steepness of the surface and highlighting changes in elevation}
}

\newglossaryentry{positiveopenness}{
    name={Positive Openness},
    description={is a terrain visualization method that emphasizes convex and elevated landforms by measuring how open the surface is upward}
}

\newglossaryentry{skyviewfactor}{
    name={Sky-View Factor},
    description={is a terrain visualization method that estimates how much of the sky is visible from each location and helps reveal subtle relief features}
}

\newglossaryentry{semanticsegmentation}{
    name={Semantic Segmentation},
    description={is a computer vision task in which each pixel of an input image or raster is assigned to a semantic class}
}

\newglossaryentry{referenceannotation}{
    name={Reference Annotation},
    description={is expert-provided spatial information used as the reference source for creating training and evaluation labels}
}

\newglossaryentry{groundtruth}{
    name={Ground Truth},
    description={is the reference data against which model predictions are compared during training or evaluation}
}

\newglossaryentry{confidenceshadow}{
    name={Confidence Shadow},
    description={is a soft-label representation in which confidence is highest inside the annotated object and gradually decreases toward the surrounding background}
}

\newglossaryentry{buffer}{
    name={Buffer},
    description={is a spatial expansion around vector geometry used to convert narrow annotations, such as terrace centre lines, into wider raster masks}
}

\newglossaryentry{validpixelmask}{
    name={Valid Pixel Mask},
    description={is a binary mask indicating which pixels contain valid data and should be included in training or metric computation}
}

\newglossaryentry{datasetmanifest}{
    name={Dataset Manifest},
    description={is a structured file describing generated tiles, their coordinates, split assignment, validity fraction, and presence of positive pixels}
}

\newglossaryentry{thresholdbasedinference}{
    name={Threshold-Based Inference},
    description={is the conversion of continuous probability predictions into binary masks using a selected probability threshold}
}

\newglossaryentry{probabilitymap}{
    name={Probability Map},
    description={is a raster output in which each pixel stores the model-estimated probability of belonging to the target class}
}

\newglossaryentry{binaryprediction}{
    name={Binary Prediction},
    description={is a thresholded model output in which each pixel is assigned either to the target class or to the background class}
}

\newglossaryentry{falsepositive}{
    name={False Positive},
    description={is a model prediction incorrectly assigned to the archaeological target class}
}

\newglossaryentry{falsenegative}{
    name={False Negative},
    description={is a reference archaeological feature or pixel that the model failed to detect}
}

\newglossaryentry{truepositive}{
    name={True Positive},
    description={is a correctly detected target pixel or object}
}
